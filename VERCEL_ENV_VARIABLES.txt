# Environment Variables for Vercel Deployment
# Copy these to Vercel Dashboard → Your Project → Settings → Environment Variables

# ==============================================
# STEP 1: Clerk Authentication (Add These First)
# ==============================================

# Clerk Publishable Key (safe for client-side)
# Variable Name: NEXT_PUBLIC_CLERK_PUBLISHABLE_KEY
# Value: Get from https://dashboard.clerk.com → API Keys
NEXT_PUBLIC_CLERK_PUBLISHABLE_KEY=pk_live_your-key-here

# Clerk Secret Key (server-side only)
# Variable Name: CLERK_SECRET_KEY
# Value: Get from https://dashboard.clerk.com → API Keys
CLERK_SECRET_KEY=sk_live_your-key-here

# Clerk Webhook Signing Secret
# Variable Name: CLERK_WEBHOOK_SIGNING_SECRET
# Value: Get from https://dashboard.clerk.com → Webhooks
CLERK_WEBHOOK_SIGNING_SECRET=whsec_your-secret-here

# Auth route configuration
NEXT_PUBLIC_CLERK_SIGN_IN_URL=/sign-in
NEXT_PUBLIC_CLERK_SIGN_UP_URL=/sign-up
NEXT_PUBLIC_CLERK_AFTER_SIGN_UP_URL=/onboarding

# ==============================================
# STEP 2: Database (Add After Setting Up Vercel Postgres)
# ==============================================

# Database Connection String
# Variable Name: DATABASE_URL
# Value: Get this from Vercel Dashboard → Storage → Postgres
# Example: postgres://default:xxxxx@ep-xxxxx.us-east-1.postgres.vercel-storage.com:5432/verceldb?sslmode=require
DATABASE_URL=postgres://...get-from-vercel-postgres...

# ==============================================
# NOTES:
# ==============================================
# 1. In Vercel, add each variable separately using the UI
# 2. Set environment to: Production, Preview, Development (all)
# 3. Google OAuth is now configured in Clerk Dashboard, not via env vars
# 4. Database setup comes in Step 5 of the deployment guide
